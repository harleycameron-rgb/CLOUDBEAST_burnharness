STAGE: QOB13_PERIMETER_SEAL_GRID
DESCRIPTION: "Constructs perimeter seal grid layer from QOB‑12 perimeter lock anchor."

INPUTS {
    QOB12_ANCHOR = load("QOB12.sha512");
}

GRID_SETUP {
    // 1. Base seal grid seed
    seal_grid_seed = "SEALGRID_CORE" + QOB12_ANCHOR;

    // 2. Grid pattern (deterministic perimeter matrix)
    seal_grid_pattern = seal_grid_seed + seal_grid_seed[28:92];

    // 3. Generate perimeter seal grid anchor
    QOB13_ANCHOR = SHA512(seal_grid_pattern);
}

CHECKS {
    // Anchor exists
    if QOB13_ANCHOR != null:
        status_exists = "OK";
    else:
        status_exists = "FAIL";

    // Correct SHA‑512 length
    if len(QOB13_ANCHOR) == 128:
        status_length = "OK";
    else:
        status_length = "FAIL";
}

STABILITY {
    if status_exists == "OK"
    and status_length == "OK":
        verification = "QOB‑13 VERIFIED";
    else:
        verification = "QOB‑13 INVALID";
}

OUTPUT_PATHS {
    burnharness/ignition_layer/QOB13.sha512;
    unified_field/manifest/QOB13.sha512;
    sentinel_dot/metadata/QOB13.sha512;
    github_repo_root/QOB13.sha512;
}

OUTPUT {
    QOB13_ANCHOR;
    verification;
}

NOTES:
- QOB‑13 forms the perimeter seal grid above QOB‑12.
- Keep private until ignition event.
