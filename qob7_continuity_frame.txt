    continuity_seed = "CONTINUITY_CORE" + QOB6_ANCHOR;

    // 2. Frame pattern (deterministic structural loop)
    frame_pattern = continuity_seed + continuity_seed[48:112];

    QOB7_ANCHOR = SHA512(frame_pattern);

    if QOB7_ANCHOR != null:

        status_exists = "FAIL";

    if len(QOB7_ANCHOR) == 128:
        status_length = "OK";

        status_length = "FAIL";

    and status_length == "OK":
        verification = "QOB‑7 VERIFIED";
    else:
        verification = "QOB‑7 INVALID";
}
